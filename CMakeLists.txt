set(opcode_cases "// Generated from the local reLCS opcode enum.\n")
foreach(line IN LISTS opcode_lines)
  if(line MATCHES "^};")
    break()
  elseif(line MATCHES "^#(if|else|endif)")
    string(APPEND opcode_cases "${line}\n")
  elseif(line MATCHES "^[ \t]*(COMMAND_[A-Z0-9_]+)")
    string(APPEND opcode_cases "case ${CMAKE_MATCH_1}: return \"${CMAKE_MATCH_1}\";\n")
  endif()
endforeach()

set(diagnostic_specs "// Source signatures for explicit diagnostic simulation only.\n")
set(audit_specs "// Primary reLCS debug signatures, used for decoding only.\n")
foreach(line IN LISTS debug_lines)
  if(line MATCHES "REGISTER_COMMAND\\((COMMAND_[A-Z0-9_]+), INPUT_ARGUMENTS\\(([^)]*)\\), OUTPUT_ARGUMENTS\\(([^)]*)\\), (true|false),")
    set(command "${CMAKE_MATCH_1}")
    set(arguments "${CMAKE_MATCH_2}")
    set(results "${CMAKE_MATCH_3}")
    set(predicate "${CMAKE_MATCH_4}")
    # PS2 retail1627 (0x2732A8) returns before CollectParameters. The debug
    # registration's one operand belongs to a different/unimplemented variant.
    if(command STREQUAL "COMMAND_1627")
      set(arguments "")
    endif()
    if(command STREQUAL "COMMAND_HAS_PAD_IN_HANDS")
      continue()
    endif()
    string(REGEX MATCHALL "ARGTYPE_[A-Z0-9_]+" args "${arguments}")
    string(REGEX MATCHALL "ARGTYPE_[A-Z0-9_]+" outs "${results}")
    list(LENGTH outs output_count)
    string(APPEND audit_specs "case ${command}: return AuditSpec{{")
    foreach(arg IN LISTS args)
      if(arg STREQUAL "ARGTYPE_STRING")
        string(APPEND audit_specs "'s',")
      elseif(arg STREQUAL "ARGTYPE_FUNCTION")
        string(APPEND audit_specs "'f',")
      elseif(arg STREQUAL "ARGTYPE_LABEL")
        string(APPEND audit_specs "'l',")
      else()
        string(APPEND audit_specs "'n',")
      endif()
    endforeach()
    string(APPEND audit_specs "},${output_count}};\n")
    if(NOT "${arguments}${results}" MATCHES "ARGTYPE_(VAR|NONE)" AND NOT command STREQUAL "COMMAND_HAS_PAD_IN_HANDS")
      string(APPEND diagnostic_specs "case ${command}: return SimSpec{{")
      foreach(arg IN LISTS args)
        if(arg STREQUAL "ARGTYPE_STRING")
          string(APPEND diagnostic_specs "true,")
        else()
          string(APPEND diagnostic_specs "false,")
        endif()
      endforeach()
      string(APPEND diagnostic_specs "},${output_count},${predicate}};\n")
    endif()
  endif()
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/diagnostic_specs.inc" "${diagnostic_specs}")

  endforeach()

    string(APPEND engine_rows "${CMAKE_MATCH_1},\n")
    math(EXPR engine_row_count "${engine_row_count}+1")
  endif()
endforeach()
